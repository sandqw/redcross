#!/bin/bash

# Set the paths and variables
input_file="valid_data.csv"     # Path to the input file
log_file="sqlldr.log"           # Path to the SQL*Loader log file
bad_file="sqlldr.bad"           # Path to the SQL*Loader bad file
control_file="control.ctl"      # Path to the SQL*Loader control file
db_username="your_username"     # Oracle database username
db_password="your_password"     # Oracle database password
db_service="your_service_name"  # Oracle database service name or SID

# Run your script to generate the input file
./your_script.sh

# Run SQL*Loader to load the file into Oracle
sqlldr "$db_username/$db_password@$db_service" control="$control_file" data="$input_file" log="$log_file" bad="$bad_file"

# Check the SQL*Loader exit code
if [ $? -eq 0 ]; then
    echo "Data loaded successfully."
else
    echo "Data loading failed. Please check the log file for details."
fi
